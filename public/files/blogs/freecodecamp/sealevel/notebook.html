def draw_sea_level_data(df: pd.DataFrame = df):
    fig, ax = plt.subplots(figsize = (8, 5))
    sns.scatterplot(data=df, x = 'Year', y = 'CSIRO Adjusted Sea Level', ax=ax)
    ax.set_xlabel('Year')
    ax.set_ylabel('Sea Level (inches)')
    ax.set_title('Rise in Sea Level')
    ax.grid(True, ls = ':')

    plt.tight_layout()
    plt.savefig('SIRO_Adjusted_Sea_Level_scatter_plot.png')

draw_sea_level_data(df=df)

def reg(df: pd.DataFrame = df):
    pred_df = df.copy()
        
    slope, intercept, r, p, se = sp.stats.linregress(pred_df['Year'], pred_df['CSIRO Adjusted Sea Level'])
    return slope, intercept

# 3.1
def sea_level_reg(df: pd.DataFrame = df):
    df = df.copy()
    
    slope, intercept = reg(df=df)
    
    fig, ax = plt.subplots(figsize = (8, 5))
    sns.scatterplot(data=df, x = 'Year', y = 'CSIRO Adjusted Sea Level', ax=ax, label = 'original data')
    ax.plot(df['Year'], intercept + slope*df['Year'], c = 'r', label = 'fitted line')
    ax.set_xlabel('Year')
    ax.set_ylabel('Sea Level (inches)')
    ax.set_title('Rise in Sea Level')
    ax.grid(True, ls = ':')
    ax.legend()

    plt.tight_layout()
    plt.savefig('SIRO_Adjusted_Sea_Level_regression_curve.png')

sea_level_reg(df=df)
